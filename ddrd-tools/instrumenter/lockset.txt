# Linux kernel lock pair file example (lockset.txt)
# 格式: <加锁函数> <解锁函数> <属性>
# 属性可自定义，如 0: mutex, 1: spinlock, 2: rwlock, 3: sem, 4: 读写信号量, 5: 其他

# mutex locks
mutex_lock mutex_unlock 0
mutex_lock_interruptible mutex_unlock 0
mutex_lock_killable mutex_unlock 0
mutex_trylock mutex_unlock 0

# spin locks
spin_lock spin_unlock 1
spin_lock_irq spin_unlock_irq 1
spin_lock_irqsave spin_unlock_irqrestore 1
spin_trylock spin_unlock 1

# read-write locks
read_lock read_unlock 2
write_lock write_unlock 2
read_lock_irq read_unlock_irq 2
write_lock_irq write_unlock_irq 2
read_lock_irqsave read_unlock_irqrestore 2
write_lock_irqsave write_unlock_irqrestore 2

# semaphores
down up 3
down_interruptible up 3
down_killable up 3
down_trylock up 3

down_read up_read 4
down_write up_write 4
down_read_trylock up_read 4
down_write_trylock up_write 4

# completion (作为同步原语)
wait_for_completion complete 5
wait_for_completion_timeout complete 5
wait_for_completion_interruptible complete 5
wait_for_completion_killable complete 5

# 其他常见锁
rwlock_init rwlock_destroy 5
mutex_init mutex_destroy 5
spin_lock_init spin_lock_destroy 5
